\section*{Theorem}

If a mould $A = (A(X), A(X_1, X_2))$ satisfies the linear double shuffle relations, then the map
\[
\begin{array}{rcl}
\mathbb{Q}[\{Z_k\}_{k\geq1}, \{Z_{i,j}\}_{i,j\geq1}] / I & \to & \mathbb{Q}[\text{coefficients}(A)] \\
Z_k & \mapsto & a(k) \\
Z_{i,j} & \mapsto & a(i,j),
\end{array}
\]
where $I$ is the ideal generated by
\[
\left( Z_{i,j} + Z_{j,i} + Z_{i+j} - \sum_{r=1}^{i+j-1} \left( \binom{r-1}{i-1} + \binom{r-1}{j-1} \right) Z_{r, i+j-r} + \delta_{i+j,2}Z(2) \right)_{i,j \geq 1}
\]
is a surjective $\mathbb{Q}$-algebra homomorphism.

\section*{Lemma 1}

\begin{enumerate}
    \item[a)] The pairing on $V_k$ given by
    \[
    \langle F(X_1, X_2), G(X_1, X_2) \rangle = \frac{1}{(k-2)!} F\left(-\frac{\partial}{\partial X_2}, \frac{\partial}{\partial X_1}\right) G(X_1, X_2) \Big|_{(0,0)}
    \]
    is bilinear, symmetric and non-degenerate.
    
    \item[b)] For all $\gamma \in \Gamma$ we have
    \[
    \langle F(X_1, X_2)|_\gamma, G(X_1, X_2)|_\gamma \rangle = \langle F(X_1, X_2), G(X_1, X_2) \rangle.
    \]
\end{enumerate}

- \textbf{the subspaces}
\begin{align*}
V_k^+ = V_k^\epsilon &= \{ p \in V_k \mid p(X_1, X_2) = p(X_2, X_1) \}, \\
V_k^- &= \{ p \in V_k \mid p(X_1, X_2) + p(X_2, X_1) = 0 \}, \\
V_k^{\text{ev}} = V_k^\delta &= \{ p \in V_k \mid p(X_1, X_2) = p(-X_1, X_2) \}, \\
V_k^{\text{odd}} &= \{ p \in V_k \mid p(X_1, X_2) + p(-X_1, X_2) = 0 \}
\end{align*}
define orthogonal decompositions $V_k \cong V_k^+ \oplus V_k^- \cong V_k^{\text{ev}} \oplus V_k^{\text{odd}}$.

- \textbf{extended pairing}
\[
\langle \cdot, \cdot \rangle : V_k \times \mathbb{Q}[[X_1, X_2]] \to \mathbb{Q}
\]

\section*{Lemma 2}
An $A$-valued mould $\mathcal{A}$ satisfies the linear double shuffle relations, if and only if
\[
A(X_1, X_2)|_\Delta = \frac{A(X_1) - A(X_2)}{X_1 - X_2} - a(2)
\]
where
\[
\Delta = T + T\epsilon - 1 - \epsilon = (T-1)(1+\epsilon) \in \mathbb{Z}[\Gamma].
\]

\section*{Lemma 3}
Let $k \geq 4$ be even. Let $\mathcal{A}$ be a generic solution to the linear double shuffle relations. Then a non-trivial linear relation in the coefficients of $\mathcal{A}$ of the form
\[
\sum_{\substack{r+s=k \\ r,s \geq 1}} \lambda_{r,s} a(r,s) = \lambda_k a(k), \quad \lambda_{r,s}, \lambda_k \in \mathbb{Q}
\]
determines a non-zero polynomial $F \in V_k^+$, such that
\[
\left\langle F(X_1, X_2), \frac{A(X_1) - A(X_2)}{X_1 - X_2} - a(2) \right\rangle = \lambda_k a(k)
\]
\[
\left\langle F(X_1, X_2)|_{\Delta^*}, A(X_1, X_2) \right\rangle = \sum_{\substack{r+s=k \\ r,s \geq 1}} \lambda_{r,s} a(r,s),
\]
where $\Delta^* = (1+\epsilon)(T^{-1}-1)$ is the adjoint to $\Delta$ with respect to the pairing $\langle \cdot, \cdot \rangle$.
Conversely, if polynomials $F, F' \in V_k^+$ yield the same relation, then $\pi^+(F) = \pi^+(F')$.

\section*{Problem}
Let $k \geq 4$ be even. Let $\mathcal{A}$ be a generic solution to the linear double shuffle relations. Then the relations
\[
\frac{1}{4}a(k) = \sum_{n=1}^{(k-2)/2} a(2n+1, k-2n+1)
\]
and
\[
a(2n, k-2n) = \sum_{\substack{r+s=k \\ r,s \text{ odd}}} \lambda_{r,s}a(r,s) + \lambda_k a(k), \quad \left(n=1 \dots \frac{k-2}{2}\right),
\]
where the coefficients are given by the homogenized Bernoulli polynomials $\beta_{k,2n}(X_1, X_2)$ via
\[
\lambda_k a(k) = \left\langle \beta_{k,2n}(X_1, X_2)\big|_{(1+\epsilon-ST)^{-1}}, \frac{A(X_1) - A(X_2)}{X_1 - X_2} - a(2) \right\rangle,
\]
\[
a(2n, k-2n) - \sum_{\substack{r+s=k \\ r,s \text{ odd}}} \lambda_{r,s}a(r,s) = \left\langle \beta_{k,2n}(X_1, X_2)\big|_{(1+\epsilon-ST)(T^{-1}-1)}, A(X_1, X_2) \right\rangle.
\]
\textbf{Determine a basis for the linear relations among the coefficients of $\mathcal{A}$.}
